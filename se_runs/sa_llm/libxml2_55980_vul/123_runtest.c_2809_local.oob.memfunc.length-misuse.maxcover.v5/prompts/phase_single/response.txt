#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

#define BUFFER_SIZE 4500

char expression[BUFFER_SIZE];
FILE *xpathOutput;
void *xmlGenericErrorContext;

void xmlGenericError(void *ctx, const char *msg, ...) {
}

int main(void) {
    char filename[256];
    char *temp;
    FILE *input;
    int len;
    
    klee_make_symbolic(filename, sizeof(filename), "filename");
    klee_assume(filename[sizeof(filename)-1] == '\0');
    
    klee_make_symbolic(expression, sizeof(expression), "expression");
    
    temp = malloc(100);
    if (temp == NULL) return 0;
    
    xpathOutput = fopen("/dev/null", "w");
    if (xpathOutput == NULL) return 0;
    
    input = fopen(filename, "rb");
    if (input == NULL) {
        free(temp);
        fclose(xpathOutput);
        return 0;
    }
    
    klee_assume(fgets(expression, BUFFER_SIZE, input) != NULL);
    
    len = strlen(expression);
    len--;
    
    while ((len >= 0) &&
           ((expression[len] == '\n') || (expression[len] == '\t') ||
            (expression[len] == '\r') || (expression[len] == ' '))) len--;
    
    SAILR_ASSERT(len >= 0 && len + 1 < BUFFER_SIZE);
    klee_assert(0 && "SAILR_REACH_ASSERT");
    
    expression[len + 1] = 0;
    
    fclose(input);
    fclose(xpathOutput);
    free(temp);
    return 0;
}
