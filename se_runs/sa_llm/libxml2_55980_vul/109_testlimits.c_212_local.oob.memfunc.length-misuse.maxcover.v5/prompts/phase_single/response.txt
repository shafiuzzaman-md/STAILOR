#include <stdlib.h>
#include <string.h>
#include <stdio.h>
#include "klee/klee.h"

#define CHUNK 4096

typedef struct {
    const char *end;
} HugeTest;

static HugeTest hugeTests[10];
static char filling[CHUNK];
static int currentTest = 0;
static int instate = 0;
static size_t curlen = 0;
static size_t maxlen = 0;
static size_t dotlen = 0;
static char buffer[CHUNK];

void testFunction(void) {
    size_t len;
    klee_make_symbolic(&len, sizeof(len), "len");
    
    if (instate == 1) {
        if (len > CHUNK) len = CHUNK;
        
        SAILR_ASSERT(len <= CHUNK);
        klee_assert(0 && "SAILR_REACH_ASSERT");
        
        memcpy(buffer, &filling[0], len);
        curlen += len;
    }
}

int main(void) {
    klee_make_symbolic(&instate, sizeof(instate), "instate");
    klee_make_symbolic(&curlen, sizeof(curlen), "curlen");
    klee_make_symbolic(&maxlen, sizeof(maxlen), "maxlen");
    klee_make_symbolic(&dotlen, sizeof(dotlen), "dotlen");
    klee_make_symbolic(&currentTest, sizeof(currentTest), "currentTest");
    
    klee_assume(instate == 1);
    klee_assume(currentTest >= 0 && currentTest < 10);
    
    for (int i = 0; i < 10; i++) {
        hugeTests[i].end = "";
    }
    
    testFunction();
    
    return 0;
}
