int xmlCheckCdataPush(xmlChar c, struct _xmlParserCtxt *ctxt);

    klee_assume(buffer_size >= 1 && buffer_size <= 1024);
    klee_assume(cur_offset >= 0 && cur_offset < buffer_size);
    klee_assume(checkIndex >= 0 && checkIndex < (int)buffer_size);
    
    /* Allocate buffer for input data */
    xmlChar *buffer = malloc(buffer_size);
    if (!buffer) {
        free(ctxt->input);
        free(ctxt);
        return 1;
    }

    klee_make_symbolic(buffer, buffer_size, "buffer");
    
    /* Set up the parser input structure */
    ctxt->input->cur = buffer + cur_offset;
    ctxt->input->end = buffer + buffer_size;
    ctxt->checkIndex = checkIndex;

    klee_assume(ctxt->input->cur <= ctxt->input->end);
    
    /* Call the target function */
    int result = xmlCheckCdataPush(c, ctxt);
    
    /* Vulnerability assertion: the length argument to memchr must be non-negative
       and cur must not exceed end. The actual condition is that:
       ctxt->input->end - cur >= 0 (which is true given our assumption above)
       and the length doesn't cause OOB read in memchr.
       
       For memchr(cur, c, ctxt->input->end - cur), we need to ensure:
       1. cur is within the buffer bounds
       2. The length doesn't exceed the remaining buffer
       3. The length is non-negative
       
       The most precise condition is that cur points to valid memory and
       the length doesn't exceed the actual allocated buffer size.
    */
    SAILR_ASSERT(ctxt->input->cur >= buffer && 
                 ctxt->input->cur <= buffer + buffer_size &&
                 (size_t)(ctxt->input->end - ctxt->input->cur) <= buffer_size - (ctxt->input->cur - buffer));
    
    /* Reachability assertion */
    klee_assert(0 && "SAILR_REACH_ASSERT");
    
    /* Cleanup */
    free(buffer);
    free(ctxt->input);
    free(ctxt);
    
    return result;
}

/* Implementation of the target function (inlined since we can't link) */
int xmlCheckCdataPush(xmlChar c, struct _xmlParserCtxt *ctxt) {
    const xmlChar *cur;
    
    if (ctxt->checkIndex == 0) {
        cur = ctxt->input->cur + 1;
    } else {
        cur = ctxt->input->cur + ctxt->checkIndex;
    }
    
    if (memchr(cur, c, ctxt->input->end - cur) == NULL) {
        ctxt->checkIndex = ctxt->input->end - ctxt->input->cur;
        return 0;
    } else {
        ctxt->checkIndex = 0;
        return 1;
    }
}
