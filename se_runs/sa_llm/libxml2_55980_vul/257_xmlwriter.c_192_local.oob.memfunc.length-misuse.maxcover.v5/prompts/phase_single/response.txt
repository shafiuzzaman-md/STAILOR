xmlTextWriterPtr xmlNewTextWriter(void) {

    ret = (xmlTextWriterPtr) xmlMalloc(sizeof(xmlTextWriter));
    if (ret == NULL) {
        xmlWriterErrMsg(NULL, XML_ERR_NO_MEMORY,
                        "xmlNewTextWriter : out of memory!\n");
        return NULL;
    }
    /* TARGET LINE 192 */
    memset(ret, 0, sizeof(xmlTextWriter));

    ret->nodes = xmlListCreate(NULL, NULL);
    if (ret->nodes == NULL) {
        xmlWriterErrMsg(NULL, XML_ERR_NO_MEMORY,
                        "xmlNewTextWriter : out of memory!\n");

int main(void) {
    /* Call the target function */
    xmlTextWriterPtr writer = xmlNewTextWriter();
    
    /* Vulnerability assertion: ensure the pointer allocated is valid
       and the size used in memset is not larger than allocated */
    if (writer != NULL) {
        /* For memset length-misuse: ensure we don't write beyond allocated bounds.
           Since xmlMalloc allocates exactly sizeof(xmlTextWriter), and memset uses
           exactly that size, the condition is that the allocation succeeded and
           the pointer is non-NULL (already checked). The vulnerability would be
           if sizeof(xmlTextWriter) exceeds the actual allocated size, but here
           allocation and memset use same size. However, the SA pattern flags
           memset with potentially unbounded length. We assert the allocation
           size is at least the memset size (always true here). */
        SAILR_ASSERT(1); /* The allocation succeeded, so memset size equals allocated size */
        
        /* Reachability marker */
        klee_assert(0 && "SAILR_REACH_ASSERT");
    }
    
    /* Cleanup */
    if (writer != NULL) {
        if (writer->nodes != NULL) {
            free(writer->nodes);
        }
        free(writer);
    }
    
    return 0;
