/* Copy of the vulnerable function from xmlreader.c */
static char *xmlTextReaderBuildMessage(const char *msg, va_list ap) {
    int size = 0;
    int chars;
    char *larger;
    char *str = NULL;
    va_list aq;

    while (1) {
        /* Target line 4654 */
        va_copy(aq, ap);
        chars = vsnprintf(str, size, msg, aq);
        va_end(aq);
        if (chars < 0) {
            xmlGenericError(NULL, "vsnprintf failed !\n");
            if (str)
                xmlFree(str);
            return NULL;
        }
        if (size > chars)
            break;
        size = chars + 1;
        larger = (char *)malloc(size);
        if (!larger) {
            if (str)
                xmlFree(str);
            return NULL;
        }
        if (str) {
            memcpy(larger, str, size - 1);
            xmlFree(str);
        }
        str = larger;
    }
    return str;
}

/* Wrapper to call xmlTextReaderBuildMessage with symbolic arguments */
static void test_xmlTextReaderBuildMessage(void) {
    char msg_buf[256];
    int arg1, arg2;
    char *arg3;

    /* Make inputs symbolic */
    klee_make_symbolic(msg_buf, sizeof(msg_buf), "msg_buf");
    klee_make_symbolic(&arg1, sizeof(arg1), "arg1");
    klee_make_symbolic(&arg2, sizeof(arg2), "arg2");
    klee_make_symbolic(&arg3, sizeof(arg3), "arg3");

    /* Ensure msg_buf is null-terminated */
    klee_assume(msg_buf[255] == '\0');

    /* Prepare va_list with symbolic arguments */
    va_list ap;
    va_start(ap, msg_buf);
    /* Simulate variadic arguments - in reality we'd push them onto ap */
    /* For KLEE, we just need to reach the target line */
    
    char *result = xmlTextReaderBuildMessage(msg_buf, ap);
    
    /* Vulnerability assertion: vsnprintf should not write beyond allocated buffer */
    /* The vulnerable pattern is length-misuse in memcpy at line 4670 (not shown in snippet) */
    /* But the SA target is line 4654, so we assert safety of the vsnprintf call */
    /* Condition: vsnprintf's size parameter should be non-negative and not cause OOB */
    SAILR_ASSERT(size >= 0 && size <= 1024); /* Reasonable bound for size */
    
    /* Reachability marker */
    klee_assert(0 && "SAILR_REACH_ASSERT");
    
    va_end(ap);
    if (result) {
        xmlFree(result);
    }
}

int main(void) {
    test_xmlTextReaderBuildMessage();
    return 0;
