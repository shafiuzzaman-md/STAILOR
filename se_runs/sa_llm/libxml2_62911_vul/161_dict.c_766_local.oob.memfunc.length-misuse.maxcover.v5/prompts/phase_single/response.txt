#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

typedef struct _xmlDict xmlDict;
typedef xmlDict* xmlDictPtr;

struct _xmlDict {
    unsigned int size;
    unsigned int nbElems;
    void* table;
};

typedef struct _xmlDictEntry xmlDictEntry;
struct _xmlDictEntry {
    const xmlChar* name;
    unsigned int len;
    unsigned int hashValue;
    xmlDictEntry* next;
};

static const xmlChar* xmlDictLookup(xmlDictPtr dict, const xmlChar* name, int len) {
    xmlDictEntry* entry;
    xmlDictEntry* cur;
    xmlDictEntry* end;
    unsigned int hashValue;
    unsigned int index;
    int i;
    
    if (dict == NULL || name == NULL) return NULL;
    
    if (len < 0) {
        len = 0;
        while (name[len] != 0) len++;
    }
    
    hashValue = 0;
    for (i = 0; i < len; i++) {
        hashValue = hashValue * 31 + name[i];
    }
    
    index = hashValue % dict->size;
    entry = (xmlDictEntry*)dict->table + index;
    
    if (entry->name == NULL) {
        entry->name = name;
        entry->len = len;
        entry->hashValue = hashValue;
        entry->next = NULL;
        dict->nbElems++;
        return entry->name;
    }
    
    cur = entry;
    while (cur != NULL) {
        if (cur->hashValue == hashValue &&
            cur->len == len &&
            memcmp(cur->name, name, len) == 0) {
            return cur->name;
        }
        cur = cur->next;
    }
    
    cur = entry;
    end = (xmlDictEntry*)dict->table + dict->size;
    
    if (cur < end) {
        if (cur->name != NULL) {
            xmlDictEntry* newEntry;
            
            if (cur + 1 < end) {
                newEntry = cur + 1;
            } else {
                newEntry = (xmlDictEntry*)dict->table;
            }
            
            if (newEntry->name == NULL) {
                memmove(&entry[1], entry, (char*)cur - (char*)entry);
                
                SAILR_ASSERT((char*)cur - (char*)entry <= (char*)end - (char*)entry);
                klee_assert(0 && "SAILR_REACH_ASSERT");
                
                entry->name = name;
                entry->len = len;
                entry->hashValue = hashValue;
                entry->next = NULL;
                dict->nbElems++;
                return entry->name;
            }
        }
    }
    
    return NULL;
}

int main(void) {
    xmlDict dict;
    xmlDictEntry* table;
    xmlChar name[256];
    int len;
    unsigned int table_size;
    
    klee_make_symbolic(&table_size, sizeof(table_size), "table_size");
    klee_assume(table_size >= 2 && table_size <= 256);
    
    table = (xmlDictEntry*)malloc(table_size * sizeof(xmlDictEntry));
    klee_assume(table != NULL);
    
    for (unsigned int i = 0; i < table_size; i++) {
        table[i].name = NULL;
        table[i].len = 0;
        table[i].hashValue = 0;
        table[i].next = NULL;
    }
    
    dict.size = table_size;
    dict.nbElems = 0;
    dict.table = table;
    
    klee_make_symbolic(name, sizeof(name), "name");
    klee_make_symbolic(&len, sizeof(len), "len");
    
    klee_assume(len >= 0 && len < 256);
    klee_assume(name[len] == 0);
    
    table[0].name = name;
    table[0].len = len;
    table[0].hashValue = 123;
    table[0].next = NULL;
    dict.nbElems = 1;
    
    xmlDictLookup(&dict, name, len);
    
    free(table);
    return 0;
}
