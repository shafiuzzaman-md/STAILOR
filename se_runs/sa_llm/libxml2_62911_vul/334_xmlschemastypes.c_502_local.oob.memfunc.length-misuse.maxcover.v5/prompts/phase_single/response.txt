#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

typedef struct _xmlSchemaWildcard xmlSchemaWildcard;
struct _xmlSchemaWildcard {
    int any;
    int processContents;
};

typedef struct _xmlSchemaType xmlSchemaType;
struct _xmlSchemaType {
    xmlSchemaWildcard *attributeWildcard;
};

xmlSchemaType *xmlSchemaTypeAnyTypeDef = NULL;

void xmlSchemaTypeErrMemory(void *ctx, const char *msg) {
}

xmlSchemaWildcard* xmlSchemaNewWildcard(void) {
    return (xmlSchemaWildcard*)malloc(sizeof(xmlSchemaWildcard));
}

xmlSchemaType* xmlSchemaInitBasicType(const char *name) {
    return (xmlSchemaType*)malloc(sizeof(xmlSchemaType));
}

void xmlSchemaInitTypes(void) {
    xmlSchemaWildcard *wild = NULL;
    wild = xmlSchemaNewWildcard();
    if (wild == NULL) {
        xmlSchemaTypeErrMemory(NULL, "could not create an attribute wildcard on anyType");
        return;
    }
    klee_make_symbolic(wild, sizeof(xmlSchemaWildcard), "wild");
    klee_assume(wild != NULL);
    int an, on;
    klee_make_symbolic(&an, sizeof(an), "an");
    klee_make_symbolic(&on, sizeof(on), "on");
    klee_assume(an >= 0);
    klee_assume(on >= 0);
    int size = sizeof(xmlSchemaWildcard);
    SAILR_ASSERT(an <= size && on <= size);
    klee_assert(0 && "SAILR_REACH_ASSERT");
    memset(wild, 0, sizeof(xmlSchemaWildcard));
    wild->any = 1;
    wild->processContents = 1;
    if (xmlSchemaTypeAnyTypeDef == NULL) {
        xmlSchemaTypeAnyTypeDef = (xmlSchemaType*)malloc(sizeof(xmlSchemaType));
        klee_assume(xmlSchemaTypeAnyTypeDef != NULL);
    }
    xmlSchemaTypeAnyTypeDef->attributeWildcard = wild;
    xmlSchemaType* xmlSchemaTypeAnySimpleTypeDef = xmlSchemaInitBasicType("anySimpleType");
}

int main(void) {
    xmlSchemaTypeAnyTypeDef = NULL;
    xmlSchemaInitTypes();
    return 0;
}
