#include "klee/klee.h"

typedef struct _xmlElementContent xmlElementContent;
typedef xmlElementContent *xmlElementContentPtr;
typedef unsigned char xmlChar;

typedef enum {
    XML_ELEMENT_CONTENT_ONCE
} xmlElementContentOcur;

typedef enum {
    XML_ELEMENT_CONTENT_PCDATA
} xmlElementContentType;

struct _xmlElementContent {
    int type;
    int ocur;
    const xmlChar *name;
    xmlElementContentPtr c1;
    xmlElementContentPtr c2;
    xmlElementContentPtr parent;
};

void *xmlMalloc(size_t size) {
    void *ptr = malloc(size);
    if (ptr) {
        klee_make_symbolic(ptr, size, "xmlMalloc_mem");
    }
    return ptr;
}

void xmlVErrMemory(void *ctx, const char *msg) {
}

xmlElementContentPtr xmlNewElementContent(const xmlChar *name, int type) {
    xmlElementContentPtr ret;
    ret = (xmlElementContentPtr) xmlMalloc(sizeof(xmlElementContent));
    if (ret == NULL) {
        xmlVErrMemory(NULL, "malloc failed");
        return(NULL);
    }
    memset(ret, 0, sizeof(xmlElementContent));
    ret->type = type;
    ret->ocur = XML_ELEMENT_CONTENT_ONCE;
    if (name != NULL) {
        int l;
        const xmlChar *tmp;
        tmp = name;
        while (*tmp != 0) {
            tmp++;
        }
        l = tmp - name;
        if (l > 0) {
            ret->name = name;
        }
    }
    return ret;
}

int main(void) {
    xmlChar name[100];
    int type;
    klee_make_symbolic(name, sizeof(name), "name");
    klee_make_symbolic(&type, sizeof(type), "type");
    klee_assume(type == XML_ELEMENT_CONTENT_PCDATA);
    for (int i = 0; i < 99; i++) {
        klee_assume(name[i] != 0);
    }
    name[99] = 0;
    xmlElementContentPtr ret = xmlNewElementContent(name, type);
    if (ret != NULL) {
        SAILR_ASSERT(0);
        klee_assert(0 && "SAILR_REACH_ASSERT");
    }
    return 0;
}
