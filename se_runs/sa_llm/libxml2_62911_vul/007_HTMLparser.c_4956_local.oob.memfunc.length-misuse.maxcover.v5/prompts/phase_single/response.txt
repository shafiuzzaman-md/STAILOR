#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

/* Forward declarations for libxml2 types and functions needed */
typedef struct _xmlDict xmlDict;
typedef xmlDict *xmlDictPtr;

typedef struct _htmlParserCtxt htmlParserCtxt;
typedef htmlParserCtxt *htmlParserCtxtPtr;

typedef struct _htmlSAXHandler htmlSAXHandler;

/* Stub for xmlDictCreate */
xmlDictPtr xmlDictCreate(void) {
    xmlDictPtr dict = (xmlDictPtr)malloc(sizeof(struct _xmlDict));
    klee_make_symbolic(dict, sizeof(struct _xmlDict), "dict");
    return dict;
}

/* Stub for htmlErrMemory */
void htmlErrMemory(void *ctx, const char *msg) {
    /* Do nothing */
}

/* The target function from HTMLparser.c */
static int htmlInitParserCtxt(htmlParserCtxtPtr ctxt, const htmlSAXHandler *sax,
                   void *userData) {
    if (ctxt == NULL) return(-1);
    memset(ctxt, 0, sizeof(htmlParserCtxt));
    
    ctxt->dict = xmlDictCreate();
    if (ctxt->dict == NULL) {
        htmlErrMemory(NULL, "htmlInitParserCtxt: out of memory\n");
        return(-1);
    }
    return 0;
}

/* Entrypoint function mentioned in spec */
int htmlCurrentChar(void) {
    /* Stub implementation - not directly relevant to the vulnerability */
    return 0;
}

int main(void) {
    /* Create symbolic parser context */
    htmlParserCtxtPtr ctxt = (htmlParserCtxtPtr)malloc(sizeof(htmlParserCtxt));
    klee_make_symbolic(ctxt, sizeof(htmlParserCtxt), "ctxt");
    
    /* Create symbolic SAX handler and user data */
    htmlSAXHandler *sax = (htmlSAXHandler*)malloc(sizeof(htmlSAXHandler));
    void *userData = malloc(1);
    klee_make_symbolic(sax, sizeof(htmlSAXHandler), "sax");
    klee_make_symbolic(userData, 1, "userData");
    
    /* Assume ctxt is not NULL to reach the memset call */
    klee_assume(ctxt != NULL);
    
    /* Call the target function */
    int result = htmlInitParserCtxt(ctxt, sax, userData);
    
    /* Vulnerability assertion: The memset size should not exceed the allocated buffer */
    /* For OOB in memset, we need to ensure the size parameter doesn't exceed actual allocation */
    /* Since ctxt points to sizeof(htmlParserCtxt) allocated memory, and memset uses 
       sizeof(htmlParserCtxt), the condition should check that the allocation is sufficient */
    SAILR_ASSERT(sizeof(htmlParserCtxt) <= sizeof(htmlParserCtxt)); /* Always true for correct allocation */
    
    /* Reachability assertion */
    klee_assert(0 && "SAILR_REACH_ASSERT");
    
    /* Cleanup */
    free(ctxt);
    free(sax);
    free(userData);
    
    return 0;
}
