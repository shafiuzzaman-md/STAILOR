#define FILENAME_BUF_SIZE 1024
#define DIR_BUF_SIZE 1024

    filename[FILENAME_BUF_SIZE - 1] = '\0';

    size_t filename_len = strlen(filename);
    klee_assume(filename_len < FILENAME_BUF_SIZE);  /* Stay within our buffer */
