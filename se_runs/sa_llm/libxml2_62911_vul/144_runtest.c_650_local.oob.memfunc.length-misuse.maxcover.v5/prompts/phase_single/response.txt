#include <klee/klee.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdlib.h>

int main(void) {
    int fd1, fd2;
    char bytes1[4096];
    ssize_t res1, res2;
    int total;

    fd1 = open("/dev/zero", O_RDONLY);
    if (fd1 < 0) {
        return -1;
    }

    fd2 = open("/dev/null", O_WRONLY);
    if (fd2 < 0) {
        close(fd1);
        return -1;
    }

    total = 0;
    do {
        res1 = read(fd1, bytes1, 4096);
        if (res1 <= 0)
            break;
        total += res1;
        res2 = write(fd2, bytes1, res1);
        if (res2 <= 0 || res2 != res1) {
            SAILR_ASSERT(res1 <= 4096);
            klee_assert(0 && "SAILR_REACH_ASSERT");
        }
    } while (1);

    close(fd1);
    close(fd2);
    return 0;
}
