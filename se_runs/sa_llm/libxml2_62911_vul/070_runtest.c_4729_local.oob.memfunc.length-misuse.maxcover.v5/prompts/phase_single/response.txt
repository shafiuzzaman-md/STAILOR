#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

#define BUFFER_SIZE 4500

int main(void) {
    char expr[BUFFER_SIZE];
    int len;
    int ret;
    
    klee_make_symbolic(expr, sizeof(expr), "expr");
    
    klee_assume(expr[0] != '#');
    
    len = strlen(expr);
    klee_assume(len >= 0);
    
    len--;
    
    while ((len >= 0) && (expr[len] == '\n' || expr[len] == '\r')) {
        expr[len] = 0;
        len--;
    }
    
    SAILR_ASSERT(len >= 0 && len < BUFFER_SIZE);
    klee_assert(0 && "SAILR_REACH_ASSERT");
    
    return 0;
}
