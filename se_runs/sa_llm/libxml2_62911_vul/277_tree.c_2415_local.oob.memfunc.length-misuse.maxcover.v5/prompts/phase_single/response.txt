#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

/* Forward declarations for libxml2 types and functions */
typedef struct _xmlNode xmlNode;
typedef xmlNode *xmlNodePtr;
typedef char xmlChar;

/* Minimal stub types */
struct _xmlNode {
    int type;
    const xmlChar *name;
    xmlChar *content;
    /* Other fields omitted for brevity */
};

/* Stub functions */
void xmlTreeErrMemory(const char *msg) {
    /* Do nothing */
}

void *xmlMalloc(size_t size) {
    return malloc(size);
}

xmlChar *xmlStrdup(const xmlChar *cur) {
    if (cur == NULL) return NULL;
    size_t len = strlen((const char *)cur) + 1;
    xmlChar *res = (xmlChar *)malloc(len);
    if (res) memcpy(res, cur, len);
    return res;
}

/* Constants */
static const xmlChar xmlStringText[] = "text";

/* Target function - simplified version of xmlStringGetNodeList */
xmlNodePtr xmlStringGetNodeList(const xmlChar *content) {
    xmlNodePtr cur;

    cur = (xmlNodePtr) xmlMalloc(sizeof(xmlNode));
    if (cur == NULL) {
        xmlTreeErrMemory("building text");
        return(NULL);
    }
    memset(cur, 0, sizeof(xmlNode));
    cur->type = 3; /* XML_TEXT_NODE */

    cur->name = xmlStringText;
    if (content != NULL) {
        cur->content = xmlStrdup(content);
    }

    return cur;
}

int main(void) {
    /* Make content symbolic to explore different paths */
    char content_buf[1024];
    klee_make_symbolic(content_buf, sizeof(content_buf), "content_buf");
    
    /* Assume reasonable bounds for symbolic content */
    klee_assume(content_buf[sizeof(content_buf)-1] == '\0');
    
    /* Call the target function */
    xmlNodePtr result = xmlStringGetNodeList((const xmlChar *)content_buf);
    
    /* Vulnerability assertion: check that xmlMalloc succeeded and 
       memset size is valid (always true for sizeof(xmlNode), but we
       assert based on the pattern's concern about unbounded length) */
    if (result != NULL) {
        /* For length-misuse pattern, we assert that the allocation size
           is non-negative and reasonable. Since sizeof(xmlNode) is fixed,
           we assert it's positive. */
        SAILR_ASSERT(sizeof(xmlNode) > 0);
        
        /* Reachability marker */
        klee_assert(0 && "SAILR_REACH_ASSERT");
    }
    
    /* Cleanup */
    if (result) {
        free(result->content);
        free(result);
    }
    
    return 0;
}
