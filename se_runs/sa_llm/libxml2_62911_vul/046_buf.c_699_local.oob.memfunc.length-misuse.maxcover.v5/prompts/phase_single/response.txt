/* Minimal definitions to compile the relevant code from buf.c */
typedef unsigned char xmlChar;
typedef struct _xmlBuf xmlBuf;
struct _xmlBuf {
    xmlChar *content;        
    xmlChar *contentIO;      
    unsigned int use;        
    unsigned int size;       
    unsigned int alloc;      
};

#define XML_BUFFER_ALLOC_IO 1

/* Stub for xmlRealloc */
void* xmlRealloc(void *ptr, size_t size) {
    if (size == 0) return NULL;
    return realloc(ptr, size);
}

void xmlBufAdd(xmlBuf *buf, const xmlChar *str, int len) {
    /* Simplified version that leads to the target code */
    if (buf == NULL) return;
    
    unsigned int newSize;
    if (len < 0) return;
    
    if (buf->use + len + 1 < buf->use) return; /* overflow check */
    newSize = buf->use + len + 1;
    
    /* This mimics the logic around the target line */
    if ((buf->alloc == XML_BUFFER_ALLOC_IO) && (buf->contentIO != NULL)) {
        int start_buf = buf->content - buf->contentIO;
        
        if (start_buf > newSize) {
            /* move data back to start - TARGET LINE 699 */
            memmove(buf->contentIO, buf->content, buf->use);
            buf->content = buf->contentIO;
            buf->content[buf->use] = 0;
            buf->size += start_buf;
            
            /* VULNERABILITY ASSERTION: memmove length should not exceed buffer bounds */
            SAILR_ASSERT(buf->use <= (buf->size - start_buf));

            klee_assert(0 && "SAILR_REACH_ASSERT");
        } else {
            /* This branch contains the xmlRealloc call at line 699 */
            xmlRealloc(buf->contentIO, start_buf + newSize);
        }
    }
}

int main(void) {
    /* Create a symbolic buffer structure */
    xmlBuf buf;

    klee_make_symbolic(&buf.alloc, sizeof(buf.alloc), "buf_alloc");

    klee_make_symbolic(&start_offset, sizeof(start_offset), "start_offset");

    klee_assume(buf.size >= contentIO_size);

    klee_make_symbolic(add_str, add_len + 1, "add_str");

    xmlBufAdd(&buf, add_str, add_len);
    
    /* Cleanup */
    free(buf.contentIO);
    free(add_str);
    
    return 0;
