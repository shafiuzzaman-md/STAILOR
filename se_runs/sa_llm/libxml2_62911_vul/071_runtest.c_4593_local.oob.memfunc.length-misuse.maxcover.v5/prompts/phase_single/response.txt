#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

#define BUFFER_SIZE 4500

int main(void) {
    char expression[BUFFER_SIZE];
    int len;
    FILE *input = stdin;
    FILE *output = stdout;
    char *temp = NULL;
    
    klee_make_symbolic(expression, sizeof(expression), "expression");
    
    klee_assume(expression[BUFFER_SIZE - 1] == '\0');
    
    if (fgets(expression, BUFFER_SIZE, input) != NULL) {
        len = strlen(expression);
        
        if (len > 0) {
            len--;
            
            while ((len >= 0) &&
                   ((expression[len] == '\n') || (expression[len] == '\t') ||
                    (expression[len] == '\r') || (expression[len] == ' '))) {
                len--;
                
                if (len < 0) {
                    break;
                }
                
                SAILR_ASSERT(len >= 0 && len < BUFFER_SIZE);
                klee_assert(0 && "SAILR_REACH_ASSERT");
            }
        }
    }
    
    if (temp != NULL) {
        free(temp);
    }
    
    return 0;
}
