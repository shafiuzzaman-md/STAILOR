#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

typedef struct _xmlDict xmlDict;
typedef xmlDict *xmlDictPtr;

typedef struct _xmlDictStrings xmlDictStrings;
typedef xmlDictStrings *xmlDictStringsPtr;

struct _xmlDictStrings {
    xmlDictStringsPtr next;
    char *free;
    char *end;
    int size;
    int nbStrings;
    char array[1];
};

struct _xmlDict {
    int ref_counter;
    xmlDictStringsPtr strings;
};

static void *xmlMalloc(size_t size) {
    return malloc(size);
}

static void *xmlRealloc(void *ptr, size_t size) {
    return realloc(ptr, size);
}

static void xmlFree(void *ptr) {
    free(ptr);
}

static xmlDictStringsPtr xmlDictStringsCreate(size_t size) {
    xmlDictStringsPtr ret;
    
    ret = (xmlDictStringsPtr) xmlMalloc(sizeof(xmlDictStrings) + size);
    if (ret == NULL) return NULL;
    
    ret->size = size;
    ret->nbStrings = 0;
    ret->free = &ret->array[0];
    ret->end = &ret->array[size];
    ret->next = NULL;
    
    return ret;
}

static const xmlChar *xmlDictLookup(xmlDictPtr dict, const xmlChar *name, int namelen) {
    xmlDictStringsPtr pool;
    const xmlChar *ret;
    const xmlChar *prefix = NULL;
    int plen = 0;
    size_t size;
    
    if (dict == NULL || name == NULL || namelen < 0) return NULL;
    
    if (dict->strings == NULL) {
        size = 4096;
        if (size < namelen + plen + 3) {
            size = namelen + plen + 3;
        }
        dict->strings = xmlDictStringsCreate(size);
        if (dict->strings == NULL) return NULL;
    }
    
    pool = dict->strings;
    
    if (pool->free + namelen + plen + 3 > pool->end) {
        size_t newsize = 2 * pool->size;
        if (newsize < namelen + plen + 3) {
            newsize = namelen + plen + 3;
        }
        
        pool = (xmlDictStringsPtr) xmlRealloc(pool, sizeof(xmlDictStrings) + newsize);
        if (pool == NULL) return NULL;
        
        pool->size = newsize;
        pool->free = &pool->array[0] + (pool->free - pool->array);
        pool->end = &pool->array[newsize];
        dict->strings = pool;
    }
    
found_pool:
    ret = pool->free;
    
    if (plen > 0) {
        memcpy(pool->free, prefix, plen);
        pool->free += plen;
        *(pool->free++) = ':';
    }
    
    memcpy(pool->free, name, namelen);
    pool->free += namelen;
    *(pool->free++) = 0;
    pool->nbStrings++;
    
    return ret;
}

int main(void) {
    xmlDict dict;
    xmlChar name[256];
    int namelen;
    const xmlChar *result;
    
    dict.ref_counter = 1;
    dict.strings = NULL;
    
    klee_make_symbolic(name, sizeof(name), "name");
    klee_make_symbolic(&namelen, sizeof(namelen), "namelen");
    
    klee_assume(namelen >= 0);
    klee_assume(namelen < 256);
    
    result = xmlDictLookup(&dict, name, namelen);
    
    if (result != NULL) {
        xmlDictStringsPtr pool = dict.strings;
        if (pool != NULL) {
            char *before_memcpy = pool->free;
            
            SAILR_ASSERT(pool->free + namelen <= pool->end);
            
            klee_assert(0 && "SAILR_REACH_ASSERT");
        }
    }
    
    if (dict.strings != NULL) {
        xmlFree(dict.strings);
    }
    
    return 0;
}
