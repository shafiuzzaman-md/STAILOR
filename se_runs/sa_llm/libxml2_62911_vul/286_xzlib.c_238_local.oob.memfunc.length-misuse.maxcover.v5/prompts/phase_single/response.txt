#include <stdint.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include "klee/klee.h"

typedef struct {
    int fd;
} xz_state;

static int read(int fd, void *buf, size_t count) {
    klee_make_symbolic(&count, sizeof(count), "read_count");
    klee_assume(count >= 0);
    
    int ret;
    klee_make_symbolic(&ret, sizeof(ret), "read_ret");
    klee_assume(ret <= (int)count);
    
    return ret;
}

int __libxml2_xzread(xz_state *state, unsigned char *buf, unsigned int len) {
    unsigned int have = 0;
    int ret = 0;
    
    if (state == NULL || buf == NULL) {
        return -1;
    }
    
    do {
        ret = read(state->fd, buf + have, len - have);
        if (ret <= 0)
            break;
        have += ret;
    } while (have < len);
    
    if (ret < 0) {
        SAILR_ASSERT(have <= len);
        klee_assert(0 && "SAILR_REACH_ASSERT");
        return ret;
    }
    
    return 0;
}

int main(void) {
    xz_state state;
    unsigned char *buf;
    unsigned int len;
    
    klee_make_symbolic(&state, sizeof(state), "state");
    klee_make_symbolic(&len, sizeof(len), "len");
    
    klee_assume(len >= 0);
    klee_assume(len < 1024);
    
    buf = (unsigned char *)malloc(len);
    klee_assume(buf != NULL);
    
    klee_make_symbolic(buf, len, "buf_data");
    
    int result = __libxml2_xzread(&state, buf, len);
    
    free(buf);
    return 0;
}
