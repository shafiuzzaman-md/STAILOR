#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

typedef int SORT_TYPE;

typedef struct {
    SORT_TYPE* storage;
    size_t size;
} TIM_SORT_STORE;

#define MIN(a, b) ((a) < (b) ? (a) : (b))

void TIM_SORT_RESIZE(TIM_SORT_STORE* store, size_t new_size) {
    if (new_size > store->size) {
        store->storage = realloc(store->storage, new_size * sizeof(SORT_TYPE));
        store->size = new_size;
    }
}

void TIM_SORT_MERGE_LO(TIM_SORT_STORE* store, SORT_TYPE* dst, size_t curr, size_t A, size_t B) {
    size_t i, j, k;
    SORT_TYPE* storage;
    
    TIM_SORT_RESIZE(store, MIN(A, B));
    storage = store->storage;
    
    if (A < B) {
        memcpy(storage, &dst[curr], A * sizeof(SORT_TYPE));
        i = 0;
        j = curr + A;
        
        for (k = curr; k < curr + A + B; k++) {
            if ((i < A) && (j < curr + A + B)) {
                SAILR_ASSERT(A <= store->size && "memcpy length exceeds allocated storage");
                klee_assert(0 && "SAILR_REACH_ASSERT");
            }
        }
    }
}

int main(void) {
    TIM_SORT_STORE store;
    SORT_TYPE* dst;
    size_t curr, A, B;
    
    store.storage = NULL;
    store.size = 0;
    
    klee_make_symbolic(&curr, sizeof(curr), "curr");
    klee_make_symbolic(&A, sizeof(A), "A");
    klee_make_symbolic(&B, sizeof(B), "B");
    
    klee_assume(A > 0);
    klee_assume(B > 0);
    klee_assume(curr >= 0);
    
    dst = malloc((curr + A + B) * sizeof(SORT_TYPE));
    klee_assume(dst != NULL);
    
    TIM_SORT_MERGE_LO(&store, dst, curr, A, B);
    
    free(store.storage);
    free(dst);
    
    return 0;
}
