#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

char* xmlFgets(char* line_read, int size, FILE* stream) {
    if (!fgets(line_read, size, stream))
        return NULL;
    line_read[size] = 0;
    return line_read;
}

char* my_prompt(const char* prompt) {
    char line_read[501];
    int len;
    char* ret;

    if (prompt != NULL)
        fprintf(stdout, "%s", prompt);
    fflush(stdout);
    
    klee_make_symbolic(line_read, sizeof(line_read), "line_read");
    klee_assume(line_read[499] == '\0');
    
    if (!xmlFgets(line_read, 500, stdin))
        return NULL;
    
    len = strlen(line_read);
    ret = (char*)malloc(len + 1);
    
    if (ret != NULL) {
        SAILR_ASSERT(len <= 500);
        klee_assert(0 && "SAILR_REACH_ASSERT");
    }
    
    return ret;
}

int testSAX(void) {
    char* result = my_prompt("test> ");
    if (result) {
        free(result);
    }
    return 0;
}

int main(void) {
    return testSAX();
}
