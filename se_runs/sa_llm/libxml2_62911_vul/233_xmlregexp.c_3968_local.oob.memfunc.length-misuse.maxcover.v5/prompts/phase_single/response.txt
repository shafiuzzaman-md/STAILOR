#include <stdlib.h>
#include <string.h>
#include "klee/klee.h"

/* Minimal stub types to compile without libxml2 headers */
typedef struct _xmlRegExecCtxt {
    char *errString;
    int state;
    int *errCounts;
    int *counts;
    struct _xmlRegComp {
        int nbCounters;
    } *comp;
} xmlRegExecCtxt;

/* Stub functions to avoid linking with libxml2 */
void xmlFree(void *ptr) {
    if (ptr) free(ptr);
}

char *xmlStrdup(const char *cur) {
    if (!cur) return NULL;
    size_t len = strlen(cur) + 1;
    char *res = (char *)malloc(len);
    if (res) memcpy(res, cur, len);
    return res;
}

/* Entrypoint from SA spec */
void xmlRegStateAddTrans(xmlRegExecCtxt *exec, const char *value) {
    if (!exec || !value) return;
    
    if (exec->errString != NULL)
        xmlFree(exec->errString);
    exec->errString = xmlStrdup(value);
    exec->errState = exec->state;
    
    /* TARGET LINE 3968: memcpy with potential OOB */
    if (exec->comp->nbCounters) {
        /* VULNERABILITY ASSERTION: ensure nbCounters doesn't cause OOB */
        SAILR_ASSERT(exec->comp->nbCounters > 0 && 
                     exec->errCounts != NULL && 
                     exec->counts != NULL);
        
        /* REACHABILITY MARKER */
        klee_assert(0 && "SAILR_REACH_ASSERT");
        
        memcpy(exec->errCounts, exec->counts,
               exec->comp->nbCounters * sizeof(int));
    }
}

int main(void) {
    /* Symbolic inputs for the execution context */
    xmlRegExecCtxt exec;
    exec.comp = (typeof(exec.comp))malloc(sizeof(*exec.comp));
    
    /* Make nbCounters symbolic - this is the critical length variable */
    klee_make_symbolic(&exec.comp->nbCounters, sizeof(exec.comp->nbCounters), "nbCounters");
    
    /* Constrain nbCounters to reasonable positive values */
    klee_assume(exec.comp->nbCounters >= 0);
    klee_assume(exec.comp->nbCounters <= 100); /* Reasonable bound */
    
    /* Allocate arrays based on symbolic nbCounters */
    if (exec.comp->nbCounters > 0) {
        exec.errCounts = (int *)malloc(exec.comp->nbCounters * sizeof(int));
        exec.counts = (int *)malloc(exec.comp->nbCounters * sizeof(int));
        
        /* Assume allocations succeeded */
        klee_assume(exec.errCounts != NULL);
        klee_assume(exec.counts != NULL);
        
        /* Initialize counts array with symbolic values */
        klee_make_symbolic(exec.counts, exec.comp->nbCounters * sizeof(int), "counts");
    } else {
        exec.errCounts = NULL;
        exec.counts = NULL;
    }
    
    /* Initialize other fields */
    exec.errString = NULL;
    exec.state = 0;
    
    /* Symbolic value parameter */
    char value[100];
    klee_make_symbolic(value, sizeof(value), "value");
    value[99] = '\0'; /* Ensure null termination */
    
    /* Call the target function */
    xmlRegStateAddTrans(&exec, value);
    
    /* Cleanup */
    if (exec.errCounts) free(exec.errCounts);
    if (exec.counts) free(exec.counts);
    if (exec.errString) free(exec.errString);
    if (exec.comp) free(exec.comp);
    
    return 0;
}
