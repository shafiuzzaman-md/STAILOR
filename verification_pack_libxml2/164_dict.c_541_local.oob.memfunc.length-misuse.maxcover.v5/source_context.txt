  516: xmlDictFindEntry(const xmlDict *dict, const xmlChar *prefix,
  517:                  const xmlChar *name, int len, unsigned hashValue,
  518:                  int *pfound) {
  519:     xmlDictEntry *entry;
  520:     unsigned mask, pos, displ;
  521:     int found = 0;
  522: 
  523:     mask = dict->size - 1;
  524:     pos = hashValue & mask;
  525:     entry = &dict->table[pos];
  526: 
  527:     if (entry->hashValue != 0) {
  528:         /*
  529:          * Robin hood hashing: abort if the displacement of the entry
  530:          * is smaller than the displacement of the key we look for.
  531:          * This also stops at the correct position when inserting.
  532:          */
  533:         displ = 0;
  534: 
  535:         do {
  536:             if (entry->hashValue == hashValue) {
  537:                 if (prefix == NULL) {
  538:                     /*
  539:                      * name is not necessarily null-terminated.
  540:                      */
  541:                     if ((memcmp(entry->name, name, len) == 0) &&
  542:                         (entry->name[len] == 0)) {
  543:                         found = 1;
  544:                         break;
  545:                     }
  546:                 } else {
  547:                     if (xmlStrQEqual(prefix, name, entry->name)) {
  548:                         found = 1;
  549:                         break;
  550:                     }
  551:                 }
  552:             }
  553: 
  554:             displ++;
  555:             pos++;
  556:             entry++;
  557:             if ((pos & mask) == 0)
  558:                 entry = dict->table;
  559:         } while ((entry->hashValue != 0) &&
  560:                  (((pos - entry->hashValue) & mask) >= displ));
  561:     }
  562: 
  563:     *pfound = found;
  564:     return(entry);
  565: }
